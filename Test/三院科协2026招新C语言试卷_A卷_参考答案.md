### 22. 限定池抽卡日志分析（10 分）



#### 1. 解题思路

1. 本题只需要读 $k$ 行出金记录，与 $n$ 的大小无关：$n$ 取到 $10^7$ 也不影响运行时间（$O(k)$），更不需要开 $n$ 大小的数组。
2. **出金次数**就是 $k$；**歪的次数**是 $r_i=2$ 的个数，边读边数即可。
3. 用一个变量 `last_p` 记住"上一次出金的位置"，初值为 0。这样每读到第 $i$ 次出金的位置 $p_i$，这一段垫抽长度就是 $p_i-\text{last\_p}-1$；第一次出金时它自然等于 $p_1-1$，首段不用单独处理。
4. 每算出一段长度就与当前最大值比较、更新**最长垫抽**；然后记下新的 `last_p`。
5. 全部读完后再算一次**尾段** $n-\text{last\_p}$：它既是**当前水位**，也要拿去更新最长垫抽（样例 2 的最长垫抽就来自结尾）。
6. 全程没出金（$k=0$）时循环一次都不执行，`last_p` 仍是 0，于是水位 $=n$、最长垫抽 $=n$，与题意一致，不需要特判。

#### 2. 参考程序

```c
#include <stdio.h>

int main(void)
{
    int n, k;
    scanf("%d %d", &n, &k);              /* n 总抽数，k 出金次数 */

    int miss = 0;                        /* 歪了几次 */
    int max_gap = 0;                     /* 最长垫抽 */
    int last_p = 0;                      /* 上一次出金的位置，0 表示还没出过金 */

    for (int i = 0; i < k; i++) {
        int p, r;
        scanf("%d %d", &p, &r);          /* p：第几抽出金；r：1 当期，2 歪 */

        int gap = p - last_p - 1;        /* 这一抽之前连续垫了多少抽 */
        if (gap > max_gap)
            max_gap = gap;
        if (r == 2)
            miss = miss + 1;

        last_p = p;
    }

    int water = n - last_p;              /* 最后一次出金之后又垫了多少抽 */
    if (water > max_gap)
        max_gap = water;                 /* 结尾这一段也要算 */

    printf("出金次数: %d 歪了: %d 次 最长垫抽: %d 当前水位: %d\n",
           k, miss, max_gap, water);

    return 0;
}
```

### 23. 排雷机器人（10 分）



#### 1. 题目本质与三种解法概览

题目要求：对每个**非雷**格子，统计其 8 邻域中雷的个数；雷格本身原样输出 `*`。

本质上就是"在二维网格上做 8 邻域统计"。常见的三种写法：

| 解法               | 核心想法                                         | 关键点                                           |
| ------------------ | ------------------------------------------------ | ------------------------------------------------ |
| 解法一：方向数组   | 对每个格子，用 `dx[8]`、`dy[8]` 枚举 8 个方向    | 方向数组要"配对"，且必须判断越界                 |
| 解法二：外圈哨兵   | 把地图放进 $(n+2)\times(m+2)$ 的数组，外圈铺 `.` | 读入从下标 1 开始，哨兵值不能用 `*`              |
| 解法三：以雷为中心 | 反过来，对每颗雷把它 8 个邻居的计数 $+1$         | 需要一个额外的计数数组，且雷格输出时不能输出计数 |

三种解法复杂度相同：时间 $O(nm)$（常数因子分别为全格 8 次判断、全格 8 次查询、雷数 ×8 次自增），空间 $O(nm)$。

#### 2. 解法一：方向数组

**思路**：把 8 个方向的偏移量预先存进两个数组，遍历每个非雷格子时按数组累加邻居中的雷数。越界用 `x >= 0 && x < n && y >= 0 && y < m` 判断。

```c
/* 解法一：方向数组
 * 对每个非雷格子，按 dx/dy 数组枚举 8 个方向，判断越界后累加相邻地雷数。 */
#include <stdio.h>

int main(void)
{
    /* 8 个方向：左上、上、右上、左、右、左下、下、右下 */
    const int dx[8] = {-1, -1, -1,  0, 0,  1, 1, 1};
    const int dy[8] = {-1,  0,  1, -1, 1, -1, 0, 1};

    char g[105][105];
    int n, m;

    if (scanf("%d %d", &n, &m) != 2)
        return 0;

    for (int i = 0; i < n; i++)
        scanf("%s", g[i]);

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (g[i][j] == '*') {
                putchar('*');
                continue;
            }

            int cnt = 0;
            for (int k = 0; k < 8; k++) {
                int x = i + dx[k];
                int y = j + dy[k];
                if (x >= 0 && x < n && y >= 0 && y < m && g[x][y] == '*')
                    cnt++;
            }
            putchar('0' + cnt);
        }
        putchar('\n');
    }

    return 0;
}
```

#### 3. 解法二：外圈哨兵

**思路**：把整个地图放进 $(n+2)\times(m+2)$ 的字符数组，最外圈全部铺成 `.`（哨兵），真正的地图从下标 $1$ 开始存。这样内部任意格子的 8 个邻居都落在数组范围内，**循环体里一句越界判断都不需要**。

```c
/* 解法二：外圈哨兵
 * 把雷区放在 (n+2) x (m+2) 的数组里，最外圈全部填成 '.'（哨兵），
 * 真正的地图从下标 1 开始存放。这样枚举 8 个邻居时永远不越界，
 * 循环体内不需要任何边界判断。 */
#include <stdio.h>

int main(void)
{
    char g[105][105];
    int n, m;

    if (scanf("%d %d", &n, &m) != 2)
        return 0;

    for (int i = 1; i <= n; i++)
        scanf("%s", &g[i][1]);          /* 第 i 行从下标 1 开始存 m 个字符 */

    /* 铺哨兵：第 0 行、第 n+1 行、第 0 列、第 m+1 列全部设为 '.' */
    for (int j = 0; j <= m + 1; j++) {
        g[0][j] = '.';
        g[n + 1][j] = '.';
    }
    for (int i = 0; i <= n + 1; i++) {
        g[i][0] = '.';
        g[i][m + 1] = '.';
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (g[i][j] == '*') {
                putchar('*');
                continue;
            }

            int cnt = 0;
            for (int di = -1; di <= 1; di++)
                for (int dj = -1; dj <= 1; dj++)
                    if (g[i + di][j + dj] == '*')   /* 无需判断越界 */
                        cnt++;                      /* (i,j) 自己不是雷，不必跳过 */

            putchar('0' + cnt);
        }
        putchar('\n');
    }

    return 0;
}
```

#### 4. 解法三：以雷为中心

**思路**：换一个视角——不再"逐格数邻居"，而是"逐雷扩散"：遍历所有格子，遇到一颗雷，就把它周围 8 个格子的计数各加 1。全部雷处理完后，再遍历一遍地图输出：雷格输出 `*`，其余格子输出计数。

```c
/* 解法三：以雷为中心
 * 换一个视角：不再"逐格数邻居"，而是"逐雷扩散"——
 * 先把 (i,j) 周围的 8 个格子计数加 1，所有雷处理完后，
 * 再遍历一遍地图按计数输出。 */
#include <stdio.h>

int main(void)
{
    char g[105][105];
    int cnt[105][105] = {0};
    int n, m;

    if (scanf("%d %d", &n, &m) != 2)
        return 0;

    for (int i = 0; i < n; i++)
        scanf("%s", g[i]);

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (g[i][j] != '*')
                continue;               /* 只处理地雷格 */

            for (int di = -1; di <= 1; di++) {
                for (int dj = -1; dj <= 1; dj++) {
                    if (di == 0 && dj == 0)
                        continue;       /* 跳过雷自己 */
                    int x = i + di, y = j + dj;
                    if (x >= 0 && x < n && y >= 0 && y < m)
                        cnt[x][y]++;    /* 这颗雷给邻居的"威胁值" +1 */
                }
            }
        }
    }

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (g[i][j] == '*')
                putchar('*');
            else
                putchar('0' + cnt[i][j]);
        }
        putchar('\n');
    }

    return 0;
}
```

#### 5. 三种解法对比

| 对比项     | 解法一 方向数组                | 解法二 外圈哨兵            | 解法三 以雷为中心                   |
| ---------- | ------------------------------ | -------------------------- | ----------------------------------- |
|            |                                |                            |                                     |
| 额外空间   | 无                             | 无（只是多铺一圈）         | 一个 `int cnt[105][105]`            |
| 越界判断   | 每个方向一次                   | 完全不需要                 | 每个邻居一次                        |
| 主要风险   | 方向漏写、方向配错             | 哨兵值误用 `*`             | 忘记清零、雷格输出计数              |
| 适用场景   | 万能模板，可复用到其它八邻域题 | 网格题通用技巧，代码最干净 | 雷稀疏时更省时间                    |
| 时间复杂度 | $O(8nm)$                       | $O(8nm)$                   | $O(8K)$（$K$ 为雷数）+ $O(nm)$ 输出 |

- 雷格输出成了数字：扣 1 分；
- 只差每行换行或每行字符数不对：扣 1 分。

### 24. 暮色大师（10 分）



#### 1. 解法一：记忆化 DP

```c
#include <stdio.h>

#define MAXN 1005

char A[MAXN], B[MAXN];
int n, m;
int cache[MAXN][MAXN];

int len(const char *str) {
    int length = 0;
    while (str[length] != '\0') {
        length++;
    }
    return length;
}

int maxInt(const int a, const int b) {
    return a > b ? a : b;
}

//dp(i, j) 表示 A[i...n-1] 和 B[j...m-1]的最长公共子序列长度。
int dp(const int i, const int j) {
    if (i == n || j == m) {
        return 0;
    }

    if (cache[i][j] != -1) {
        return cache[i][j];
    }

    if (A[i] == B[j]) {
        cache[i][j] = dp(i + 1, j + 1) + 1;
    } else {
        cache[i][j] = maxInt(dp(i + 1, j),dp(i, j + 1));
    }

    return cache[i][j];
}

int main(void) {
    scanf("%s", A);
    scanf("%s", B);

    n = len(A);
    m = len(B);

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cache[i][j] = -1;
        }
    }

    const int lcs = dp(0, 0);
    const int answer = n + m - 2 * lcs;

    printf("%d\n", answer);

    return 0;
}
```

```text
时间复杂度：O(nm)
空间复杂度：O(nm)
递归栈空间：O(n + m)
```

#### 2. 解法二：递推 DP（空间优化）

```c
#include <stdio.h>

#define MAXN 1005

char A[MAXN], B[MAXN];
int n, m;
int dp[MAXN];

int len(const char *str) {
    int length = 0;
    while (str[length] != '\0') {
        length++;
    }
    return length;
}

int maxInt(const int a, const int b) {
    return a > b ? a : b;
}

int main(void) {
    scanf("%s", A);
    scanf("%s", B);

    n = len(A);
    m = len(B);

    for (int i = 1; i <= n; i++) {
        int prev = 0;

        for (int j = 1; j <= m; j++) {
            const int last = dp[j];

            if (A[i - 1] == B[j - 1]) {
                dp[j] = prev + 1;
            } else {
                dp[j] = maxInt(dp[j], dp[j - 1]);
            }

            prev = last;
        }
    }

    const int lcs = dp[m];
    const int answer = n + m - 2 * lcs;

    printf("%d\n", answer);

    return 0;
}
```

```text
时间复杂度：O(nm)
空间复杂度：O(m)
```
