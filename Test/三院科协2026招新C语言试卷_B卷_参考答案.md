### 13 答案 A、B、D

### 14 答案 A、C

各选项独立考虑，均从 `a=0、b=2` 出发。A：`&&` 左侧为假，右侧不求值，表达式为 `0`，`b` 不变。B：`||` 左侧为假，需要计算 `++b`，`b` 变为 `3`，但逻辑表达式的结果是 `1`，不是 `3`。C：`||` 左侧为真，右侧不求值，`a` 不变。D：`&&` 左侧为真，需要计算 `++a`，`a` 变为 `1`，因此 D 错误。



### 15 答案 A、C

### 22. 赛博功德计数器（10 分）



#### 参考代码

```c
#include <stdio.h>

int main(void)
{
    int n;
    scanf("%d", &n);

    int total = 0;      /* 总功德 */
    int crit = 0;       /* 暴击次数 */

    for (int i = 0; i < n; i++) {
        int w;
        scanf("%d", &w);

        int hit7 = 0;           /* 是否含有数字 7 */
        int t = w;
        while (t > 0) {
            if (t % 10 == 7) {
                hit7 = 1;
                break;
            }
            t /= 10;
        }

        if (hit7 || w % 7 == 0) {   /* 撞 7 触发 AI 暴击 */
            total += 2 * w;
            crit++;
        } else {
            total += w;
        }
    }

    printf("%d\n%d\n", total, crit);
    return 0;
}
```

#### 样例验证

| 输入                | 期望输出   | 实测输出     |
| ------------------- | ---------- | ------------ |
| `5` / `7 3 14 5 17` | `84` `3`   | `84` `3` ✓   |
| `3` / `770 13 999`  | `2552` `1` | `2552` `1` ✓ |

#### 给分标准（共 10 分）

- 正确读入 $n$ 与 $n$ 个力度值：**2 分**
- 循环逐个处理，正确判断"7 的倍数"（`w % 7 == 0`）：**2 分**
- 正确进行数位拆分判断"含有数字 7"（`%10` 取位 + `/10` 前进）：**3 分**
- 暴击翻倍与普通结算逻辑正确（注意两个条件用 `||` 连接，只翻倍一次）：**2 分**
- 按格式输出总功德与暴击次数：**1 分**

扣分规则：语法错误一个扣 1 分，最多扣 3 分；其他错误（如条件写成 `&&`、数位拆分后忘记 `break` 导致重复计数、输出格式错误等）一个扣 1 分。

#### 常见错误

1. 判断含 7 时只判断个位，或忘记对多层数位继续 `w /= 10`；
2. 把"含 7 **或** 7 的倍数"写成 `&&`，导致样例 1 中力度 17 不触发暴击；
3. 暴击时功德加了两次 `w` 却忘了 `crit++`，或反之；
4. `while` 拆分中忘记 `break`/`else` 结构导致重复累加。

### 23. AI 主播的抽免单活动（10 分）



#### 参考代码（数组标记模拟）

```c
#include <stdio.h>

int main(void)
{
    int n, m;
    scanf("%d %d", &n, &m);

    int out[1001] = {0};    /* out[i] = 1 表示编号 i 的观众已被移出抽奖圈 */
    int alive = n;          /* 圈内剩余观众数 */
    int cnt = 0;            /* 当前这一轮报到的数 */
    int first = 1;

    int i = 1;              /* 从 1 号观众开始报数 */
    while (alive > 1) {
        if (!out[i]) {
            cnt++;
            if (cnt == m) {         /* 报到 m：本轮被淘汰 */
                out[i] = 1;
                alive--;
                cnt = 0;
                if (!first)
                    printf(" ");
                printf("%d", i);
                first = 0;
            }
        }
        i = i % n + 1;      /* 环形前进，走完 n 号回到 1 号 */
    }
    printf("\n");

    for (i = 1; i <= n; i++) {      /* 圈内剩下的最后一位 */
        if (!out[i]) {
            printf("%d\n", i);
            break;
        }
    }
    return 0;
}
```

#### 样例验证

| 输入  | 期望输出            | 实测输出              |
| ----- | ------------------- | --------------------- |
| `5 3` | `3 1 5 2` / `4`     | `3 1 5 2` / `4` ✓     |
| `6 2` | `2 4 6 3 1` / `5`   | `2 4 6 3 1` / `5` ✓   |
| `7 5` | `5 3 2 4 7 1` / `6` | `5 3 2 4 7 1` / `6` ✓ |

#### 备选解法一：约瑟夫递推公式（O(n)）

设 $f(1) = 0$，$f(k) = (f(k-1) + m) \bmod k$ 表示"幸存者在 $k$ 人圈中的下标（从 0 起）"，则答案为 $f(n) + 1$。
缺点：只能求出最终幸存者，**无法输出淘汰顺序**，因此使用该解法时淘汰顺序部分按完成度酌情给分。

```c
#include <stdio.h>

int main(void)
{
    int n, m;
    scanf("%d %d", &n, &m);

    int f = 0;                      /* 1 人圈中幸存者下标为 0 */
    for (int k = 2; k <= n; k++) {
        f = (f + m) % k;            /* 约瑟夫递推 */
    }
    /* 幸存者编号为 f + 1；淘汰顺序需另用数组模拟得出 */
    return 0;
}
```

#### 备选解法二：循环链表 / 双数组轮转

用循环链表（自建结构体 + 指针）或"下一个未淘汰的人"数组 `next[]` 同样可解，属于约瑟夫环的常规变体，正确即可给分。

#### 给分标准（共 10 分）

- 正确读入 $n, m$：**1 分**
- 建立淘汰状态标记（数组或链表），并正确实现"报数到 $m$ 淘汰、下一位从 1 重新报数"的环形模拟：**6 分**
  - 能正确环形回绕（取模或 `% n + 1`）而不错位：核心，占 3 分
  - 已淘汰观众不再参与报数：占 2 分
  - 报数计数器在淘汰后清零：占 1 分
- 按顺序正确输出 $n-1$ 个淘汰编号：**2 分**
- 正确输出最终幸存者编号：**1 分**

扣分规则：语法错误一个扣 1 分，最多扣 3 分；其他错误一个扣 1 分。若采用递推公式只输出幸存者，得算法部分 5 分（环形模拟 6 分按"递推正确"折算），淘汰顺序 2 分与读入 1 分视实际完成度给分，合计不超过 7 分。

#### 常见错误

1. 淘汰后没有把报数计数器清零，导致下一轮起点报数错误；
2. 环形前进写成 `i++`，走完 n 号后数组越界或漏掉 1 号；
3. 已淘汰的观众仍然参与报数（忘记 `if (!out[i])` 的判断）；
4. 循环终止条件写成 `cnt > n` 或 `alive > 0`，多淘汰一人；
5. 输出格式：淘汰编号之间多输出或漏输出空格、行末多余空格。

### 24. 棱镜室（10 分）



#### 解法一： DFS深搜

```c
/*
  prism room
  DFS + memorization
 */

#include <stdio.h>
#include <limits.h>
#include <string.h>

#define MAX_SIZE 100

char room[MAX_SIZE][MAX_SIZE]; // room[h][w]

// (可选) 记忆化优化，s[i][j][d]记录目前从d方向上经过(i, j)位置最小的转向次数
int s[MAX_SIZE][MAX_SIZE][4];
/*
  若没有记忆化，使用visited数组避免路径重叠
  char visited[MAX_SIZE][MAX_SIZE] = { 0 }; // 记录经过路径 
 */

// 0~3方向上的位移向量(结合方向计算下一步位置)
const int di[4] = { 0,  0,  1, -1 };
const int dj[4] = { 1, -1,  0,  0 }; 

int w, h;
int ci = -1, cj = -1; // 发现的第一个标记点的位置
int ans = INT_MAX;

void read(void) { 
	scanf("%d %d", &w, &h);
	getchar(); // 去除回车

	int i, j;
	/* 也可以直接当作字符串读取scanf("%s") */
	for (i = 0; i < h; i++) { // 获取输入
		for (j = 0; j < w; j++) {
			room[i][j] = getchar();
			if (room[i][j] == 'C' && ci == -1) {
				ci = i; cj = j;
				room[i][j] = '*'; // 标记为障碍物，避免重复经过起点
			}
		}
		getchar(); // 去除回车
	}
}

/* 深度优先搜索，位置(i, j)处, 累计c次转弯，d方向(从0到3)
   标志点位置出发d除值为-1，为保证不被算入一次转弯，并且避免被记忆化（因为不会再经过此标记点位置） */
void dfs(int i, int j, int c, int d) {
	if (d != -1) { 
        // 记忆化
		if (s[i][j][d] <= c) return ; // 跳过更长路径
		s[i][j][d] = c; // 更新最短路径
	}
	
	if (room[i][j] == 'C') {
		if (ans > c) ans = c;
		return ;
	}

	int x, y; //  i, j

	// 优化，优先保持原方向行进
	if (d != -1) { 
		x = i + di[d]; y = j + dj[d];
		if (x < h && x >= 0 && y < w && y >= 0
			//&& !visited[x][y]
			&& room[x][y] != '*') {
			//visited[x][y] = 1;
			dfs(x, y, c, d);
			//visited[x][y] = 0;
		}
	}

	for (int k = 0; k <= 3; k++) { // 转向的情况
		if (k == d) continue;
		x = i + di[k]; y = j + dj[k];
		if (x < h && x >= 0 && y < w && y >= 0
			//&& !visited[x][y]
			&& room[x][y] != '*') {
			//visited[x][y] = 1;
			dfs(x, y, c + (d != -1), k); // d=-1第一步不算入转向
			//visited[x][y] = 0;
		}
	}
}

int main(void)
{
	read();

	memset(s, 0x3f, sizeof(s)); // 初始化记忆序列
		
	dfs(ci, cj, 0, -1);

	printf("%d", ans);

	return 0;
}
```

#### 解法二： BFS广搜

```c
/*
  prism room
  BFS
 */

#include <stdio.h>
#include <limits.h>
#include <stdlib.h>
#include <string.h>

#define MAX_SIZE 100

char room[MAX_SIZE][MAX_SIZE]; // room[h][w]

/* 队列实现不作要求 建议用定长数组替代 */
typedef struct POS_STRUCT {
    int i, j;
    int d, c;
    struct POS_STRUCT *next;
} POS;

POS q_head = { .next = NULL };
POS * const top = &q_head;
POS *bottom = top;

/*
  这里使用记忆化s避免重复，也可以使用visited，但空间占用更大
  记录经过路径,visited[i][j][d][c]位置(i, j)处d方向c累积转向次数
  因为每次迭代一步，需要避免的是同转向次数下的重复计算，故多一维[c]
  char visited[MAX_SIZE][MAX_SIZE][4][MAX_SIZE*MAX_SIZE] = { 0 };
 */
int s[MAX_SIZE][MAX_SIZE][4] = { 0 };   // 记录i,j,d位置目前的最少转向次数

// 0~3方向上的方向向量(结合方向计算下一步位置)
const int di[4] = { 0,  0,  1, -1 };
const int dj[4] = { 1, -1,  0,  0 };

int w, h;
int ci = -1, cj = -1; // 发现的第一个标记点的位置
int ci1, cj1;   // 第二个标记点的位置
int ans = INT_MAX;

int q_empty(void) {
    return !top->next;
}

void q_push(int i, int j, int d, int c) {
    if (q_empty())
        bottom = top;
    bottom->next = (POS *)malloc(sizeof (POS));
    bottom = bottom->next;
    bottom->next = NULL;
    bottom->i = i, bottom->j = j;
    bottom->d = d, bottom->c = c;
}

POS q_pop(void) {
    POS dot;
	if (! q_empty()) {
		POS *p = top->next;
		top->next = p->next;
		dot.i = p->i, dot.j = p->j;
		dot.d = p->d, dot.c = p->c;
		free(p);
	}
    return dot;
}

void read(void) {
    scanf("%d %d", &w, &h);
    getchar(); // 去除回车

    int i, j;
    /* 也可以直接当作字符串读取scanf("%s") */
    for (i = 0; i < h; i++) { // 获取输入
        for (j = 0; j < w; j++) {
            room[i][j] = getchar();
            if (room[i][j] == 'C') {
                if (ci == -1) {
                    ci = i; cj = j;
                    room[i][j] = '*'; // 标记为障碍物，避免重复经过起点
                } else {
                    ci1 = i; cj1 = j;
                }
            }
        }
        getchar(); // 去除回车
    }
}

int valid(int i, int j) {
    return (i < h && i >= 0 && j < w && j >= 0);
}

void bfs(int i, int j) {
    POS dot;
    int x, y, c;

    for (int k = 0; k < 4; k++) { // 初始化搜索队列
        q_push(i, j, k, 0);
        //visited[i][j][k][0] = 1;
    }

    while (! q_empty()) {
        dot = q_pop();
        for (int k = 0; k < 4; k++) {
            x = dot.i + di[k];
            y = dot.j + dj[k];
            c = (k == dot.d ? 0 : 1) + dot.c;
            if (valid(x, y) /* && !visited[x][y][k][c] */ && room[x][y] != '*') {
                if (c >= s[x][y][k]) continue;
                s[x][y][k] = c;
                
                if (x == ci1 && y == cj1) { // 到达终点
                    if (ans > c) ans = c; // 更新答案
                    continue; // 跳过进一步搜索
                }

                //visited[x][y][k][c] = 1;
                q_push(x, y, k, c);
            }
        }
    }
}

int main(void)
{
    read();
    
    memset(s, 0x3f, sizeof(s));
    bfs(ci, cj);

    printf("%d", ans);

    return 0;
}
```
